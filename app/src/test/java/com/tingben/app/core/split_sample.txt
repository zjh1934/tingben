/**
*R25·直接朗读的断句：**只按符号断，绝不按字符从半句腰斩**。
*
*移植来源（现成源码，非手搓）：
*-antioch-audiobook`src/script/chunker.rs`（MIT）：`split_into_segments`先按段落、
*再按句子边界切，合并到目标长度；`is_heading_like`让短标题单独成段不与正文连读；
*`group_by_speaker`的「说话人变化必为分界」在这里等价于「句子边界必为分界」。
*-ReaderVoice-Mobile`data-room/.../semantic/SemanticSegmenter.kt`：引号栈+
*source-exact区间切片，不在引号中间断，切片拼回去逐字无损。
*-ensemble-reads`player/SegmentMerger.kt`：`MAX_SEG_CHARS`相邻合并；注释明确
*「单段本身超过上限的长独白保留原样，避免切碎语义」——所以本实现**没有字符硬切**。
*
*中文适配：句末集合加了。！？；…与”」』；段落边界\n\n；英文.!?沿用上游判据。
*不删一个字：所有空白都留在相邻段的尾部，拼回去与原文严格相等（charsIn==charsOut）。
*/

/**合成目标段长；只在句子/段落边界合并，不是硬上限。*/
constvalTARGET_CHARS=110

/**超过这个长度的长句也保留整句，不再按逗号切（见SegmentMerger的语义优先原则）。*/
constvalSOFT_CAP=260

privateconstvalOPENERS="“「『‘"
privateconstvalCLOSERS="”」』’\"'"
privateconstvalTERMINALS="。！？!?；;…"
privateconstvalSECONDARY="，、,：:"
privateconstvalAFTER_TERMINAL="。！？!?；;….\"”」』’"

dataclassStats(
valsegments:Int,
valsentenceCuts:Int,
valsoftCuts:Int,
valhardCuts:Int,
valoversize:Int,
vallongest:Int,
valendsAtPunct:Int,
valcharsIn:Int,
valcharsOut:Int
){
/**真机日志里一行就能看出「是不是又从中间断了」。*/
funsummary():String=
"段数=$segments·句/段边界=$sentenceCuts·次级边界=$softCuts·"+
"字符硬切=$hardCuts·超长整句保留=$oversize·最长段=$longest字·"+
"标点收尾=$endsAtPunct/$segments·原文=$charsIn字拼回=$charsOut字"+
if(charsIn==charsOut)"·逐字无损"else"·⚠️字数不一致"
}

/**返回「段列表+统计」。段内容含原文空白，顺序与原文完全一致。*/
funsplit(text:String):Pair<List<String>,Stats>{
if(text.isEmpty())returnemptyList<String>()toStats(0,0,0,0,0,0,0,0,0)

valn=text.length
valprimary=mutableListOf<Int>()//一级切点：句末标点/段落边界（exclusive）
varquoteDepth=0
vari=0
while(i<n){
valc=text[i]
if(cinOPENERS){quoteDepth++;i++;continue}
if(cinCLOSERS){if(quoteDepth>0)quoteDepth--;i++;continue}

//段落边界：连续两个换行。标题后面常常没有句号，必须单独断。

varj=i+2
while(j<n&&text[j].isWhitespace())j++
if(j<n)primary.add(j)
i=j
continue
}
//句末标点：引号闭合后才允许断（SemanticSegmenter的quotestack）。
if(cinTERMINALS&&quoteDepth==0){
varj=i+1
//把紧随其后的省略号、闭合引号、空白一起归到本段，切点落在下一个实字。
while(j<n&&(text[j]inAFTER_TERMINAL||text[j].isWhitespace()))j++
if(j<n)primary.add(j)
i=j
continue
}
i++
}

//一级切片：严格substring，不trim，不丢字符。
valunits=mutableListOf<String>()
varprev=0
for(binprimary.distinct().sorted()){
if(b>prev&&b<=n){units.add(text.substring(prev,b));prev=b}
}
if(prev<n)units.add(text.substring(prev,n))

//二级：太长的一级单元按次级标点（逗号/顿号/冒号）再断一次，仍然只按符号断。
varsoftCuts=0
valpieces=mutableListOf<String>()
for(uinunits){
if(u.length<=SOFT_CAP){pieces.add(u);continue}
valcuts=secondaryCuts(u)
if(cuts.isEmpty()){pieces.add(u);continue}
softCuts+=cuts.size
varp=0
for(bincuts){if(b>p)pieces.add(u.substring(p,b));p=b}
if(p<u.length)pieces.add(u.substring(p))
}

//合并到目标长度：只在piece边界合并，标题单独成段（Antiochis_heading_like）。
valout=mutableListOf<String>()
valcur=StringBuilder()
valheadingUnits=mutableListOf<String>()
for(sinpieces){
valheading=isHeadingLike(s)
if(cur.isNotEmpty()&&(cur.length+s.length>TARGET_CHARS||heading)){
out.add(cur.toString());cur.setLength(0)
}
cur.append(s)
if(heading&&cur.isNotEmpty()){
out.add(cur.toString());cur.setLength(0);headingUnits.add(s.trim())
}
}
if(cur.isNotEmpty())out.add(cur.toString())

valcharsOut=out.sumOf{it.length}
vallongest=out.maxOfOrNull{it.length}?:0
valoversize=out.count{it.length>SOFT_CAP}
valendsAtPunct=out.count{seg->
valt=seg.trimEnd()
t.isEmpty()||t.last()inTERMINALS||t.last()inCLOSERS
}
returnouttoStats(
segments=out.size,sentenceCuts=primary.size,softCuts=softCuts,
hardCuts=0,oversize=oversize,longest=longest,endsAtPunct=endsAtPunct,
charsIn=n,charsOut=charsOut
)
}

/**长单元内部的次级标点切点，切点仍落在标点之后的下一个实字。*/
privatefunsecondaryCuts(u:String):List<Int>{
valcuts=mutableListOf<Int>()
vardepth=0
varlast=0
for(iinu.indices){
valc=u[i]
if(cinOPENERS){depth++;continue}
if(cinCLOSERS){if(depth>0)depth--;continue}
if(cinSECONDARY&&depth==0&&i-last>=40){
varj=i+1
while(j<u.length&&u[j].isWhitespace())j++
if(j<u.length){cuts.add(j);last=j}
}
}
returncuts
}

/**短、无句末标点=标题，不与正文连读（抄chunker.rs::is_heading_like）。*/
privatefunisHeadingLike(s:String):Boolean{
valt=s.trim()
if(t.isEmpty()||t.length>=120)returnfalse
vallast=t.last()
returnlast!inTERMINALS&&last!inCLOSERS&&!t.endsWith("…")
}
}
